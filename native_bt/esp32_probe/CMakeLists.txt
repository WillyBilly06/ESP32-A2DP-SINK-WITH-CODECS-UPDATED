# Generic SDK flash targets would also write bootloader/table/default app. The
# owned workflow admits a specific existing OTA slot and preserves those regions.
idf_build_get_property(guard_python PYTHON)
add_custom_target(native_bt_sdk_flash_guard
    COMMAND ${guard_python} "${CMAKE_CURRENT_LIST_DIR}/../tools/reject_sdk_flash.py" VERBATIM)
foreach(unsafe_target flash app-flash bootloader-flash partition-table-flash encrypted-flash)
    if(TARGET ${unsafe_target})
        add_dependencies(${unsafe_target} native_bt_sdk_flash_guard)
    endif()
endforeach()
