cmake_minimum_required(VERSION 3.16)
option(DBT_PROBE_LDAC_BUILD "Build standalone LDAC decoder" OFF)
option(DBT_PROBE_AAC_BUILD "Build guarded standalone AAC-LC decoder" OFF)
option(DBT_PROBE_APTX_BUILD "Build standalone aptX decoder" OFF)
option(DBT_PROBE_OPUS_BUILD "Build guarded standalone Opus decoder" OFF)
option(DBT_PROBE_SMP "Attach owned LE security manager" OFF)
option(DBT_PROBE_AVRCP "Attach owned AVRCP control and metadata" OFF)
option(DBT_PROBE_LHDC_BUILD "Also compile the separate user-owned LHDC decoder component" OFF)
option(DBT_PROBE_SBC_BUILD "Compile only the SDK standalone SBC codec and owned wrapper" OFF)
option(DBT_PROBE_AUDIO_TEST "Enable encrypted AVDTP/SBC-to-PCM counters, without I2S output" OFF)
option(DBT_PROBE_KEEP_OWNED_ACTIVE "Keep owned development image selected after basic boot qualification" OFF)
option(DBT_PROBE_AUDIO_OUTPUT "Enable experimental native integer DSP/I2S and Audio Sink SDP" OFF)
option(DBT_PROBE_GATT_TEST "Enable the diagnostic unbonded LE echo/counter service" OFF)
option(DBT_PROBE_CLASSIC_SECURITY_TEST "Enable peer-restricted Classic pairing/bond diagnostics" OFF)
option(DBT_PROBE_INGRESS_TRACE "Enable bounded diagnostic ACL ingress snapshots" OFF)
option(DBT_PROBE_DISABLE_RX_FLOW "Diagnostic A/B only: leave receive credits disabled" OFF)
if(DBT_PROBE_AUDIO_TEST AND (NOT DBT_PROBE_SBC_BUILD OR NOT DBT_PROBE_CLASSIC_SECURITY_TEST))
    message(FATAL_ERROR "SBC diagnostic requires standalone codec and restricted Classic security")
endif()
if(DBT_PROBE_AUDIO_OUTPUT AND NOT DBT_PROBE_AUDIO_TEST)
    message(FATAL_ERROR "Native I2S output requires the restricted SBC audio candidate")
endif()
set(COMPONENTS main)
foreach(codec AAC APTX LDAC)
    if(DBT_PROBE_${codec}_BUILD)
        string(TOLOWER ${codec} codec_name)
        list(APPEND COMPONENTS ${codec_name}_decoder)
    endif()
endforeach()
if(DBT_PROBE_OPUS_BUILD)
    list(APPEND COMPONENTS opus_decoder)
endif()
if(DBT_PROBE_SBC_BUILD)
    list(APPEND COMPONENTS sbc_decoder)
endif()
if(DBT_PROBE_LHDC_BUILD)
    list(APPEND COMPONENTS lhdcv5_decoder)
endif()
if(DBT_PROBE_AUDIO_OUTPUT)
    list(APPEND COMPONENTS esp_driver_i2s)
endif()
set(DBT_PROBE_PAIRING_PEER_BYTES "0,0,0,0,0,0" CACHE STRING "Explicit test peer BD_ADDR in HCI byte order")
if(NOT DEFINED ENV{IDF_PATH})
    get_filename_component(local_idf "${CMAKE_CURRENT_LIST_DIR}/../../esp-idf" ABSOLUTE)
    set(ENV{IDF_PATH} "${local_idf}")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(native_bt_probe)
