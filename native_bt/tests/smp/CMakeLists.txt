cmake_minimum_required(VERSION 3.22)
project(dbt_smp_probe_psa_checks C)
get_filename_component(native_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(workspace "${native_root}/.." ABSOLUTE)
set(DBT_MBEDTLS_ROOT "${workspace}/esp-idf/components/mbedtls/mbedtls" CACHE PATH "SDK Mbed TLS 4 source")
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(USE_SHARED_TF_PSA_CRYPTO_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_TF_PSA_CRYPTO_LIBRARY ON CACHE BOOL "" FORCE)
set(TF_PSA_CRYPTO_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(TF_PSA_CRYPTO_CONFIG_FILE "${native_root}/tests/crypto_psa/config.h" CACHE FILEPATH "" FORCE)
add_subdirectory("${DBT_MBEDTLS_ROOT}/tf-psa-crypto" psa)
set(owned_sources)
foreach(unit queue hci host acl l2cap credit ertm ertm_engine channels rx_flow runtime
             smp smp_crypto smp_bond smp_guard smp_dist smp_features smp_sc smp_bearer smp_session)
    list(APPEND owned_sources "${native_root}/src/dbt_${unit}.c")
endforeach()
foreach(fixture probe_adapter probe_reconnect)
    add_executable(${fixture}_test "${CMAKE_CURRENT_LIST_DIR}/${fixture}.c" ${owned_sources})
    if(fixture STREQUAL "probe_reconnect")
        target_sources(${fixture}_test PRIVATE "${native_root}/src/esp32/dbt_smp_nvs.c" "${CMAKE_CURRENT_LIST_DIR}/nvs_mock.c")
    endif()
    target_include_directories(${fixture}_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/probe_shim"
        "${native_root}/include" "${DBT_MBEDTLS_ROOT}/include")
    target_compile_features(${fixture}_test PRIVATE c_std_11)
    # These fixtures use assert as their oracle; Release must not erase checks.
    if(MSVC)
        target_compile_options(${fixture}_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(${fixture}_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    endif()
    target_link_libraries(${fixture}_test PRIVATE tfpsacrypto)
endforeach()
